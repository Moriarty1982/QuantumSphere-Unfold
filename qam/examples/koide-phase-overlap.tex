\[
\qamformula{koide-overlap-input}{conditional}

\[
\qamformula[koide-overlap-input]{koide-overlap-moment}{conditional}
 {Z=\sum_j w_j e^{i\phi_j/3},\qquad
 x_k=\sum_j w_j x_{jk}
 =a[1+\sqrt{2}\operatorname{Re}(e^{2\pi i k/3}Z)]}
\]
\[
\qamformula[koide-overlap-moment]{koide-overlap-output}{conditional}
 {m_k=x_k^2,\qquad K=\frac{1+|Z|^2}{3},\qquad
 \phi_{\rm eff}=3\arg Z\pmod{2\pi}\quad(Z\ne0)}
\]
\[
\qamformula[koide-overlap-moment]{koide-overlap-mirror}{conditional}
 {\phi_\pm=\phi_0\pm3h,\quad w_+=w_-=1/2,\quad
 Z=e^{i\phi_0/3}\cos h,\quad |h|<\pi/2}
\]
\[
\qamformula[koide-overlap-input]{koide-overlap-incoherent-control}{conditional}
 {\sum_jw_j x_{jk}^2-\left(\sum_jw_j x_{jk}\right)^2
 =\sum_j w_j\left(x_{jk}-\sum_lw_lx_{lk}\right)^2}
\]
\end{qamcontext}
